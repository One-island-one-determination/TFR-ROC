defined_text = {
    name = GetShanheSelectedName
    
    # 當變數為 1 時，回傳山西的文本代號
    text = {
        trigger = { check_variable = { CHI_shanhe_dec_visible = 1 } }
        localization_key = shanhe_name_shanxi
    }
    # 當變數為 2 時，回傳山東的文本代號
    text = {
        trigger = { check_variable = { CHI_shanhe_dec_visible = 2 } }
        localization_key = shanhe_name_shandong
    }
    # 當變數為 3 時，回傳河南的文本代號
    text = {
        trigger = { check_variable = { CHI_shanhe_dec_visible = 3 } }
        localization_key = shanhe_name_henan
    }
    # 當變數為 4 時，回傳河北的文本代號
    text = {
        trigger = { check_variable = { CHI_shanhe_dec_visible = 4 } }
        localization_key = shanhe_name_hebei
    }
    # 當變數為 0 (未選擇任何省份) 時的防呆顯示
    text = {
        trigger = { check_variable = { CHI_shanhe_dec_visible = 0 } }
        localization_key = shanhe_name_none
    }
}
