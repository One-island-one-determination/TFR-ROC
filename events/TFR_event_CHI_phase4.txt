      attacker = {
        # Old owner attacks from their territory bordering Tuva
        state = 575
        num_provinces = 4

      defender = {
        # ROC defends Tuva
        state = 329
        num_provinces = 4
        on_win = CHI.H4.101
        on_lose = CHI.H4.102
        on_cancel = CHI.H4.103
      }
    }
  }
}
