PRC_Infiltration_change = {
  custom_effect_tooltip = PRC_Infiltration_change_tooltip
  # 進度累加
  add_to_variable = {
    var = CHI_PRC_Infiltration_progress
    value = CHI_PRC_Infiltration_progress_temp
  }
  clamp_variable = {
    var = CHI_PRC_Infiltration_progress
    min = 0
    max = 100
  }
  # 用 number 當計算用變數
  set_variable = {
    var = number
    value = CHI_PRC_Infiltration_progress_temp
  }
  divide_variable = {
    var = number
    value = 1000
  }
  multiply_variable = {
    var = number
    value = -1
  }
  # 累加 buff
  add_to_variable = {
    var = CHI_PRC_Infiltration_buff
    value = number
  }
  clamp_variable = {
    var = CHI_PRC_Infiltration_buff
    min = -0.1
    max = 0
  }
  if = {
    limit = {
      check_variable = { CHI_PRC_Infiltration_progress < 10 }
    }
    hidden_effect = {
      remove_ideas = CHI_PRC_infiltration_low
      remove_ideas = CHI_PRC_infiltration_middle
      remove_ideas = CHI_PRC_infiltration_high
      remove_ideas = CHI_People_surrender
      remove_ideas = CHI_Information_transparency
    }
  }
  else_if = {
    limit = {
      check_variable = { CHI_PRC_Infiltration_progress > 10 }
      check_variable = { CHI_PRC_Infiltration_progress < 50 }
    }
    hidden_effect = {
      add_ideas = CHI_PRC_infiltration_low
      remove_ideas = CHI_PRC_infiltration_middle
      remove_ideas = CHI_PRC_infiltration_high
      remove_ideas = CHI_People_surrender
      remove_ideas = CHI_Information_transparency
    }
  }
  else_if = {
    limit = {
      check_variable = { CHI_PRC_Infiltration_progress > 50 }
      check_variable = { CHI_PRC_Infiltration_progress < 99 }
    }
    hidden_effect = {
      remove_ideas = CHI_PRC_infiltration_low
      add_ideas = CHI_PRC_infiltration_middle
      remove_ideas = CHI_PRC_infiltration_high
      remove_ideas = CHI_People_surrender
      remove_ideas = CHI_Information_transparency
    }
  }
  else_if = {
    limit = { check_variable = { CHI_PRC_Infiltration_progress = 100 } }
    hidden_effect = {
      remove_ideas = CHI_PRC_infiltration_low
      remove_ideas = CHI_PRC_infiltration_middle
      add_ideas = CHI_PRC_infiltration_high
      add_ideas = CHI_People_surrender
      add_ideas = CHI_Information_transparency
    }
  }
}

CHI_Vote_counting_Legislator = {
  set_variable = { CHI_Green_supprt = party_popularity@social_liberal }
  set_variable = { CHI_Green_Legislator = CHI_Green_supprt }
  multiply_variable = { CHI_Green_Legislator = CHI_all_Legislator }
  round_variable = CHI_Green_Legislator

  set_variable = { CHI_Blue_supprt = party_popularity@conservative }
  add_to_variable = { CHI_Blue_supprt = party_popularity@Kuomintang2 }
  set_variable = { CHI_Blue_Legislator = CHI_Blue_supprt }
  multiply_variable = { CHI_Blue_Legislator = CHI_all_Legislator }
  round_variable = CHI_Blue_Legislator

  set_variable = { CHI_Middle_Legislator = CHI_all_Legislator }
  subtract_from_variable = { CHI_Middle_Legislator = CHI_Blue_Legislator }
  subtract_from_variable = { CHI_Middle_Legislator = CHI_Green_Legislator }

  if = {
    limit = { has_country_flag = KMT_win_flag }
    set_variable = { CHI_Base_support_Legislator = CHI_Blue_Legislator }
  }
  else_if = {
    limit = { has_country_flag = DPP_win_flag }
    set_variable = { CHI_Base_support_Legislator = CHI_Green_Legislator }
  }
  else = {
    if = {
      limit = { has_government = social_liberal }
      set_variable = { CHI_Base_support_Legislator = CHI_Green_Legislator }
    }
    else = {
      set_variable = { CHI_Base_support_Legislator = CHI_Blue_Legislator }
    }
  }

  set_variable = { CHI_governing_ratio = CHI_Base_support_Legislator }
  divide_variable = { CHI_governing_ratio = CHI_all_Legislator }
  set_variable = { CHI_governing_progress = CHI_governing_ratio }
  multiply_variable = { CHI_governing_progress = 100 }

  # Threshold indicators: 1 = grey, 2 = glowing green
  if = {
    limit = { check_variable = { CHI_Base_support_Legislator > 37 } }
    set_variable = { CHI_threshold_1_3_frame = 2 }
  }
  else = {
    set_variable = { CHI_threshold_1_3_frame = 1 }
  }

  if = {
    limit = { check_variable = { CHI_Base_support_Legislator > 56 } }
    set_variable = { CHI_threshold_1_2_frame = 2 }
  }
  else = {
    set_variable = { CHI_threshold_1_2_frame = 1 }
  }

  if = {
    limit = { check_variable = { CHI_Base_support_Legislator > 75 } }
    set_variable = { CHI_threshold_2_3_frame = 2 }
  }
  else = {
    set_variable = { CHI_threshold_2_3_frame = 1 }
  }

  CHI_update_parliament_diagram = yes
}
