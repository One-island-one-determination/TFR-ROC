scripted_gui = {
	TFR_CHI_pathguides_open = {
		window_name = "TFR_CHI_pathguides_open"
		context_type = player_context
		parent_window_token = politics_tab
		visible = {
			tag = CHI
		}
		effects = {
			TFR_CHI_pathguides_button_click = {
				if = {
					limit = {
						NOT = {
							has_country_flag = TFR_CHI_pathguides_open_flag
						}
					}
					set_country_flag = TFR_CHI_pathguides_open_flag
				}
				else = {
					clr_country_flag = TFR_CHI_pathguides_open_flag
				}
				if = {
					limit = {
						NOT = {
							OR = {
								check_variable = { var_TFR_CHI_pathguides_content_page = 1 }
								check_variable = { var_TFR_CHI_pathguides_content_page = 3 }
								check_variable = { var_TFR_CHI_pathguides_content_page = 2 }
							}
						}
					}
					set_variable = { var_TFR_CHI_pathguides_content_page = 1 }
				}
			}
		}
	}

	TFR_CHI_pathguides_main = {
		context_type = player_context

		window_name = "TFR_CHI_pathguides_main"
		ai_enabled = { always = no }

		visible = {
			has_country_flag = TFR_CHI_pathguides_open_flag
		}

		triggers = {
			#
		}

		effects = {
			TFR_CHI_pathguides_button_close_click = {
				clr_country_flag = TFR_CHI_pathguides_open_flag
			}

			TFR_CHI_pathguides_button_next_click = {
				if = {
					limit = { check_variable = { var_TFR_CHI_pathguides_content_page > 2 } }
					set_variable = { var_TFR_CHI_pathguides_content_page = 3 }
				}
				else = { add_to_variable = { var_TFR_CHI_pathguides_content_page = 1 } }
			}

			TFR_CHI_pathguides_button_prev_click = {
				if = {
					limit = { check_variable = { var_TFR_CHI_pathguides_content_page < 2 } }
					set_variable = { var_TFR_CHI_pathguides_content_page = 1 }
				}
				else = { add_to_variable = { var_TFR_CHI_pathguides_content_page = -1 } }
			}

		}

		properties = {
			TFR_CHI_pathguides_playerflag = {
				image = "[THIS.GetFlag]"
			}
		}
	}
}
