          set_variable = { CHI_PRC_Infiltration_progress_temp = 5 }

          set_variable = { CHI_Assist_reconstruction_progress_temp = 0 }
          ARR_reconstruction = yes

            var = CHI_influence_KMT_temp
            value = 0.3
          }
          CHI_influence_KMT_add = yes

            var = CHI_influence_DPP_temp

  on_weekly = {

          has_active_mission = War_weariness_mission

        }
        country_event = taiwan.34
      }
      if = {
        limit = {
          tag = CHI

          character = CHI_han_kuo_yu_char
